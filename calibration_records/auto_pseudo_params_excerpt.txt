@dataclass
class DetectParams:


    # 灰度异常阈值

    k_sigma: float = 3.0


    # 最小面积比例

    min_area_ratio: float = 0.0015


    # 最大长宽比

    max_aspect: float = 5.0


    # 综合质量阈值

    score_threshold: float = 0.55



    # 框扩展

    expand_ratio: float = 0.05



    # 高斯

    blur_size: int = 5



    # 形态学

    morph_size: int = 3



    # 稳定性

    delta_k: float = 0.25



    # 五维权重


    w_contrast: float = 0.30

    w_peak: float = 0.25

    w_stability: float = 0.20

    w_edge: float = 0.15

    w_fill: float = 0.10
